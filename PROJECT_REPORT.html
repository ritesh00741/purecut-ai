1. Executive Summary & Problem Statement

PureCut AI is a high-performance web platform that automates complex background removal for portraits, eCommerce merchandise, vehicles, and pets. Traditional thresholding or edge-detection techniques fail when dealing with complex hair strands, transparent glass, or subjects wearing clothes matching background tones.

PureCut AI resolves this through deep learning: deploying ISNet (Interactive Segmentation Network) via ONNX Runtime and Transformers.js, augmented by proprietary Hermite Smoothstep Alpha Refinement to guarantee lossless 4-channel PNG outputs with zero background haze.

Core Architectural Feat: Resolved Vercel serverless native binary stripping via Next.js outputFileTracingIncludes, allowing Linux C++ ONNX binaries to run within a 60s serverless lambda execution envelope.

2. Deep Learning Matting Pipeline

Standard neural segmentation networks produce raw sigmoid probabilities from 0.0 to 1.0. Directly applying these masks causes two major visual flaws:

  1. Sigmoid Background Noise: Residual probability values (1–15 out of 255) create faint dirty halos over dark backdrops.
  2. Foreground Incomplete Density: Values saturating at 250–254 leave subtle transparency holes inside solid subjects.

PureCut AI implements a specialized alpha transfer curve across all 4-channel pixel outputs: 

// Hermite Smoothstep Alpha Refinement Algorithm
function refineAlphaData(maskData) {
  const refined = new Uint8Array(maskData.length);
  const lowThresh = 12;   // Noise cutoff: any alpha <= 12 becomes 0 (pure transparent)
  const highThresh = 235; // Solid cutoff: any alpha >= 235 becomes 255 (solid foreground)
  const range = highThresh - lowThresh;

  for (let i = 0; i < maskData.length; i++) {
    const v = maskData[i];
    if (v <= lowThresh) {
      refined[i] = 0;
    } else if (v >= highThresh) {
      refined[i] = 255;
    } else {
      // Smooth Hermite interpolation for natural hair/fur antialiasing
      const t = (v - lowThresh) / range;
      refined[i] = Math.round(t * t * (3 - 2 * t) * 255);
    }
  }
  return refined;
}

4. Production Server Route (`app/api/process-image/route.ts`)

import { NextRequest, NextResponse } from 'next/server';

export const dynamic = 'force-dynamic';
export const maxDuration = 60;

export async function POST(req: NextRequest) {
  const startTime = Date.now();
  const formData = await req.formData();
  const file = formData.get('image') as File | null;

  if (!file || !file.type.startsWith('image/')) {
    return NextResponse.json({ error: 'Valid image required.' }, { status: 400 });
  }

  const arrayBuffer = await file.arrayBuffer();
  const inputBuffer = Buffer.from(arrayBuffer);

  const sharp = (await import('sharp')).default;
  let neuralBuffer = inputBuffer;
  const meta = await sharp(inputBuffer).metadata();
  if ((meta.width && meta.width > 1600) || (meta.height && meta.height > 1600)) {
    neuralBuffer = await sharp(inputBuffer)
      .resize(1600, 1600, { fit: 'inside', withoutEnlargement: true })
      .toBuffer();
  }

  const { RawImage, pipeline, env } = await import('@huggingface/transformers');
  env.cacheDir = process.env.VERCEL ? '/tmp/.hf-cache' : './.hf-cache';
  
  const pipe = await pipeline('image-segmentation', 'onnx-community/ISNet-ONNX');
  const rawInput = await RawImage.fromBlob(new Blob([neuralBuffer]));
  const out = await pipe(rawInput);
  
  const refinedMask = new RawImage(refineAlphaData(out[0].mask.data), out[0].mask.width, out[0].mask.height, 1);
  const cutout = rawInput.clone();
  cutout.putAlpha(refinedMask);

  const outputBuffer = await sharp(Buffer.from(cutout.data.buffer), {
    raw: { width: cutout.width, height: cutout.height, channels: 4 }
  }).png({ compressionLevel: 6 }).toBuffer();

  return new NextResponse(new Uint8Array(outputBuffer), {
    status: 200,
    headers: {
      'Content-Type': 'image/png',
      'X-PureCut-Duration-Ms': String(Date.now() - startTime),
      'X-PureCut-Model': 'ISNet-Universal-MultiObject-Refined',
    }
  });
}
